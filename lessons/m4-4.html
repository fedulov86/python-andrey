37from turtle import *

speed(0)

def move(x, y):
    penup()
    goto(x, y)
    pendown()

def square(size):
    for i in range(4):
        forward(size)
        left(90)

def triangle(size):
    for i in range(3):
        forward(size)
        left(120)

def house(x, y, size):
    move(x, y)
    color("brown")
    square(size)
    move(x, y + size)
    color("red")
    triangle(size)

def sun(x, y):
    move(x, y)
    dot(80, "gold")

house(-200, -100, 100)
house(0, -100, 60)
sun(200, 150)

hideturtle()
done()

Посмотри: функция house внутри себя вызывает другие функции — move, square, triangle. Так и строят большие программы: из маленьких понятных кусочков.
