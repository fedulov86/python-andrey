36import pgzrun
import random

WIDTH = 600
HEIGHT = 400

hero = Actor("hero", (300, 360))
rocks = []

def add_rock():
    rock = Actor("rock", (random.randint(20, WIDTH - 20), -20))
    rocks.append(rock)

def update():
    if keyboard.left:
        hero.x -= 6
    if keyboard.right:
        hero.x += 6

    for rock in rocks:
        rock.y += 4
        if rock.colliderect(hero):
            print("Попал!")

    for rock in rocks[:]:
        if rock.y > HEIGHT + 20:
            rocks.remove(rock)

def draw():
    screen.fill("midnightblue")
    hero.draw()
    for rock in rocks:
        rock.draw()

clock.schedule_interval(add_rock, 1.0)
pgzrun.go()

Что тут нового

  • Каждую секунду add_rock создаёт новый камень над экраном и кладёт его в список.
  • В update цикл двигает каждый камень вниз и проверяет, не задел ли он героя.
  • Камни, которые улетели вниз, убираем из списка, иначе их станет тысячи и игра начнёт тормозить.
  • rocks[:] — это копия списка. Удалять элементы из списка, по которому идёт цикл, нельзя: Python запутается. Поэтому идём по копии, а удаляем из настоящего.
